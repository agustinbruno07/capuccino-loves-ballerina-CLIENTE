import com.capuccinolovesballerina.game.objetos.BloqueEspuma;

public class BallerinaCapuccina extends Player {

    private BloqueEspuma espuma = null;

    private float cooldownEspuma = 0f;

    public BallerinaCapuccina(float x, float y) {
        super(x, y, Constantes.ANCHO_PERSONAJE, Constantes.ALTO_PERSONAJE);
    }

    public boolean crearEspuma() {
        if (cooldownEspuma > 0f) {
            return false;
        }

        espuma = null;

        float ancho = Constantes.ANCHO_ESPUMA;
        float alto = Constantes.ALTO_ESPUMA;
        float gap = Constantes.DISTANCIA_ESPUMA;

        float xEspuma;
        if (isMirandoDerecha()) {
            xEspuma = getX() + getAncho() + gap;
        } else {
            xEspuma = getX() - gap - ancho;
        }

        float yEspuma;
        if(enSuelo) {
            yEspuma = getY();
        }else{
            yEspuma = getY() - alto;
        }

        espuma = new BloqueEspuma(xEspuma, yEspuma, ancho, alto, Constantes.VIDA_ESPUMA);
        cooldownEspuma = Constantes.COOLDOWN_ESPUMA;
        return true;
    }

    public void actualizarHabilidad(float delta) {
        if (cooldownEspuma > 0f) {
            cooldownEspuma -= delta;
            if (cooldownEspuma < 0f) cooldownEspuma = 0f;
        }

        if (espuma != null) {
            espuma.actualizar(delta);
            if (!espuma.estaViva()) {
                espuma = null;
            }
        }
    }

    public BloqueEspuma getEspuma() {
        return espuma;
    }

    public void reiniciarHabilidad() {
        espuma = null;
        cooldownEspuma = 0f;
    }

    public void setEspumaRemota(float x, float y) {
        espuma = new BloqueEspuma(
            x,
            y,
            Constantes.ANCHO_ESPUMA,
            Constantes.ALTO_ESPUMA,
            Constantes.VIDA_ESPUMA
        );
        cooldownEspuma = 0f;
    }
}

package com.capuccinolovesballerina.game.entidades;

import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.utils.Array;
import com.capuccinolovesballerina.game.Utilidades.Constantes;

public class Player {


    protected float x;
    protected float y;
    protected float ancho;
    protected float alto;

    protected float velocidadX;
    protected float velocidadY;

    protected boolean enSuelo;
    protected boolean mirandoDerecha;

    private final Rectangle caja = new Rectangle();

    public Player() {
        this(0, 0, Constantes.ANCHO_PERSONAJE, Constantes.ALTO_PERSONAJE);
    }

    public Player(float x, float y, float ancho, float alto) {
        this.x = x;
        this.y = y;
        this.ancho = ancho;
        this.alto = alto;
        this.mirandoDerecha = true;
    }

    public void actualizar(float delta, boolean izquierda, boolean derecha, boolean saltar, Array<Rectangle> solidos) {

        float direccion = 0;

        if (izquierda) {
            direccion = -1;
        }

        if (derecha) {
            direccion = 1;
        }

        velocidadX = direccion * Constantes.VELOCIDAD_MOVIMIENTO;

        if (direccion > 0) {
            mirandoDerecha = true;
        }

        if (direccion < 0) {
            mirandoDerecha = false;
        }

        if (saltar && enSuelo) {
            velocidadY = Constantes.VELOCIDAD_SALTO;
            enSuelo = false;
        }

        velocidadY -= Constantes.GRAVEDAD * delta;

        if (velocidadY < -Constantes.VELOCIDAD_MAX_CAIDA) {
            velocidadY = -Constantes.VELOCIDAD_MAX_CAIDA;
        }

        moverEnX(delta, solidos);
        moverEnY(delta, solidos);
    }

    private void moverEnX(float delta, Array<Rectangle> solidos) {

        x += velocidadX * delta;
        caja.set(x, y, ancho, alto);

        for (Rectangle solido : solidos) {
            if (caja.overlaps(solido)) {

                if (velocidadX > 0) {
                    x = solido.x - ancho;
                } else if (velocidadX < 0) {
                    x = solido.x + solido.width;
                }

                caja.set(x, y, ancho, alto);
                velocidadX = 0;
            }
        }
    }

    private void moverEnY(float delta, Array<Rectangle> solidos) {

        y += velocidadY * delta;
        caja.set(x, y, ancho, alto);

        enSuelo = false;

        for (Rectangle solido : solidos) {
            if (caja.overlaps(solido)) {

                if (velocidadY < 0) {
                    y = solido.y + solido.height;
                    enSuelo = true;
                } else if (velocidadY > 0) {
                    y = solido.y - alto;
                }

                caja.set(x, y, ancho, alto);
                velocidadY = 0;
            }
        }
    }

    public Rectangle getCaja() {
        caja.set(x, y, ancho, alto);
        return caja;
    }

    public float getX() {
        return x;
    }

    public float getY() {
        return y;
    }

    public float getAncho() {
        return ancho;
    }

    public float getAlto() {
        return alto;
    }

    public boolean isEnSuelo() {
        return enSuelo;
    }

    public boolean isMirandoDerecha() {
        return mirandoDerecha;
    }

    public void setX(float x) {
        this.x = x;
    }

    public void setY(float y) {
        this.y = y;
    }

    public void reaparecer(float x, float y) {
        this.x = x;
        this.y = y;
        this.velocidadX = 0;
        this.velocidadY = 0;
        this.enSuelo = false;
    }

    public void setPosicionRemota(float x, float y, boolean mirandoDerecha, boolean enSuelo) {
        this.x = x;
        this.y = y;
        this.velocidadX = 0f;
        this.velocidadY = 0f;
        this.mirandoDerecha = mirandoDerecha;
        this.enSuelo = enSuelo;
    }
}

    private final Animation<TextureRegion> idle;
    private final Animation<TextureRegion> walk;
    private final Animation<TextureRegion> jump;

        textura = new Texture("sprites/ballerina.png");

        TextureRegion[][] grilla = TextureRegion.split(textura, FRAME, FRAME);

        idle = new Animation<>(0.2f, primeros(grilla[0], FRAMES_IDLE));
        idle.setPlayMode(Animation.PlayMode.LOOP);

        walk = new Animation<>(0.1f, primeros(grilla[1], FRAMES_WALK));
        walk.setPlayMode(Animation.PlayMode.LOOP);

        jump = new Animation<>(0.15f, primeros(grilla[2], FRAMES_JUMP));
        jump.setPlayMode(Animation.PlayMode.NORMAL);

        ability = new Animation<>(0.08f, primeros(grilla[3], FRAMES_ABILITY));
        ability.setPlayMode(Animation.PlayMode.NORMAL);
    }

        TextureRegion[] frames = new TextureRegion[n];
        for (int i = 0; i < n; i++) {
            frames[i] = fila[i];
        }
        return frames;
    }

    public void setEstado(Estado nuevo) {
        if (estado == Estado.ABILITY && !ability.isAnimationFinished(tiempo)) return;

        if (nuevo != estado) {
            estado = nuevo;
            tiempo = 0f;
        }
    }

    public void actualizar(float delta) {
        tiempo += delta;
    }

    public void reiniciar() {
        estado = Estado.IDLE;
        tiempo = 0f;
    }

    public TextureRegion getFrameActual() {
        switch (estado) {
            case WALK:
                return walk.getKeyFrame(tiempo);
            case JUMP:
                return jump.getKeyFrame(tiempo);
            case ABILITY:
                return ability.getKeyFrame(tiempo);
            default:
                return idle.getKeyFrame(tiempo);
        }
    }

    public void dispose() {
        textura.dispose();
    }
}

package com.capuccinolovesballerina.game.mapa;

import com.badlogic.gdx.files.FileHandle;
import com.badlogic.gdx.Files.FileType;
import com.badlogic.gdx.graphics.g2d.TextureRegion;
import com.badlogic.gdx.maps.ImageResolver;
import com.badlogic.gdx.maps.tiled.*;
import com.badlogic.gdx.utils.Array;
import com.badlogic.gdx.utils.XmlReader.Element;

/** Usa el lector TMX de LibGDX y sus mismas colisiones, sin cargar PNG ni OpenGL. */
public class CargadorMapaServidor extends TmxMapLoader {
    public CargadorMapaServidor() {
        super(ruta -> {
            FileHandle archivo = new FileHandle(ruta);
            if (archivo.exists()) return archivo;
            archivo = new FileHandle("assets/" + ruta);
            if (archivo.exists()) return archivo;
            // El servidor tambien debe arrancar desde el JAR, sin depender de la carpeta actual.
            return new FileHandle(ruta, FileType.Classpath) {};
        });
    }

    @Override
    public TiledMap load(String ruta) {
        FileHandle archivo = resolve(ruta);
        root = xml.parse(archivo);
        return loadTiledMap(archivo, new Parameters(), nombre -> new TextureRegion());
    }

    @Override
    protected void addStaticTiles(FileHandle archivo, ImageResolver imagenes, TiledMapTileSet tiles,
        Element elemento, Array<Element> elementos, String nombre, int primerId, int ancho, int alto,
        int separacion, int margen, int offsetX, int offsetY, String fuente, int anchoImagen,
        int altoImagen, FileHandle imagen) {
        int cantidad = elemento.getIntAttribute("tilecount", 0);
        if (cantidad == 0 && imagen != null) {
            cantidad = ((anchoImagen - 2 * margen + separacion) / (ancho + separacion))
                * ((altoImagen - 2 * margen + separacion) / (alto + separacion));
        }
        for (int i = 0; i < cantidad; i++) {
            // Solo importan las medidas para interpretar los objetos del tileset.
            TextureRegion region = new TextureRegion() {
                @Override public int getRegionWidth() { return ancho; }
                @Override public int getRegionHeight() { return alto; }
            };
            addStaticTiledMapTile(tiles, region, primerId + i, offsetX, offsetY);
        }
    }
}

package com.capuccinolovesballerina.game.mapa;

import com.badlogic.gdx.maps.MapObject;
import com.badlogic.gdx.maps.objects.RectangleMapObject;
import com.badlogic.gdx.maps.tiled.TiledMap;
import com.badlogic.gdx.maps.tiled.TiledMapTile;
import com.badlogic.gdx.maps.tiled.TiledMapTileLayer;
import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.utils.Array;


public class GestorColisiones {

    private final Array<Rectangle> solidos = new Array<>();

    /**
     * @param tamanoTile tamano del tile en pixeles de Tiled (por ejemplo 32)
     * @param escala     factor para pasar de pixeles de Tiled a unidades del mundo (Nivel.ESCALA)
     */
    public GestorColisiones(TiledMap mapa, int tamanoTile, float escala) {
        for (TiledMapTileLayer capa : mapa.getLayers().getByType(TiledMapTileLayer.class)) {
            if (capa.getName().toLowerCase().startsWith("fondo")) continue;
            for (int x = 0; x < capa.getWidth(); x++) {
                for (int y = 0; y < capa.getHeight(); y++) {
                    TiledMapTileLayer.Cell celda = capa.getCell(x, y);
                    if (celda == null) continue; // celda vacia

                    TiledMapTile tile = celda.getTile();

                    for (MapObject obj : tile.getObjects()) {
                        if (obj instanceof RectangleMapObject) {
                            Rectangle r = ((RectangleMapObject) obj).getRectangle();

                            // Se calcula en pixeles de Tiled y despues se escala todo junto.
                            float mundoX = x * tamanoTile + r.x;
                            float mundoY = y * tamanoTile + (tamanoTile - r.y - r.height);
                            solidos.add(new Rectangle(
                                mundoX * escala,
                                mundoY * escala,
                                r.width * escala,
                                r.height * escala
                            ));
                        }
                    }
                }
            }
        }
    }

    public Array<Rectangle> getSolidos() {
        return solidos;
    }

    public boolean colisiona(Rectangle entidad) {
        for (Rectangle r : solidos) {
            if (r.overlaps(entidad)) return true;
        }
        return false;
    }
}

package com.capuccinolovesballerina.game.mapa;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.maps.MapLayer;
import com.badlogic.gdx.maps.MapObject;
import com.badlogic.gdx.maps.MapProperties;
import com.badlogic.gdx.maps.objects.RectangleMapObject;
import com.badlogic.gdx.maps.tiled.TiledMap;
import com.badlogic.gdx.maps.tiled.TmxMapLoader;
import com.badlogic.gdx.maps.tiled.objects.TiledMapTileMapObject;
import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.utils.Array;

public class Nivel {

    /**
     * Escala entre "pixeles de Tiled" y "unidades del mundo".
     * 1f = el mundo mide lo mismo que el mapa en Tiled (30x20 tiles de 32 = 960x640).
     * Es como esta disenado NIVEL 0 (puerta 96x128, peligro 64x32, etc.).
     * Si algun dia queres un mundo mas grande sin retocar el mapa, cambia SOLO este numero.
     */
    public static final float ESCALA = 1f;

    /** true = los tiles se dibujan sin suavizar (pixel art). */
    private static final boolean PIXEL_ART = false;

    /** Dejalo en true hasta que todo funcione; despues ponelo en false. */
    private static final boolean DEBUG_OBJETOS = true;

    private final TiledMap mapa;
    private final GestorColisiones colisiones;
    private final Array<MapObject> objetos = new Array<>();
    private final float anchoPx;
    private final float altoPx;

    public Nivel(String ruta) {
        this(ruta, false);
    }

    public Nivel(String ruta, boolean servidor) {
        TmxMapLoader.Parameters parametros = new TmxMapLoader.Parameters();
        if (PIXEL_ART) {
            parametros.generateMipMaps = false;
            parametros.textureMinFilter = Texture.TextureFilter.Nearest;
            parametros.textureMagFilter = Texture.TextureFilter.Nearest;
        }
        mapa = servidor ? new CargadorMapaServidor().load(ruta) : new TmxMapLoader().load(ruta, parametros);

        MapProperties props = mapa.getProperties();
        int tileAncho = props.get("tilewidth", Integer.class);
        int tileAlto = props.get("tileheight", Integer.class);
        int tilesX = props.get("width", Integer.class);
        int tilesY = props.get("height", Integer.class);

        anchoPx = tilesX * tileAncho * ESCALA;
        altoPx = tilesY * tileAlto * ESCALA;

        // Se leen los objetos de TODAS las capas de objetos, se llamen como se llamen
        // (Tiled las crea como "Capa de Objetos 1" por defecto).
        for (MapLayer capa : mapa.getLayers()) {
            for (MapObject obj : capa.getObjects()) {
                objetos.add(obj);
            }
        }

        colisiones = new GestorColisiones(mapa, tileAncho, ESCALA);

        if (DEBUG_OBJETOS && Gdx.app != null) {
            logObjetos();
        }
    }

    public TiledMap getMapa() {
        return mapa;
    }

    public GestorColisiones getColisiones() {
        return colisiones;
    }

    /** Ancho total del mapa en unidades del mundo. */
    public float getAnchoPx() {
        return anchoPx;
    }

    /** Alto total del mapa en unidades del mundo. */
    public float getAltoPx() {
        return altoPx;
    }

    /** Primer objeto con ese nombre/type/class, en unidades del mundo (o null). */
    public Rectangle buscarObjeto(String tipo) {
        for (MapObject obj : objetos) {
            if (!coincide(obj, tipo)) continue;
            Rectangle r = rectanguloDe(obj);
            if (r != null) return r;
        }
        if (DEBUG_OBJETOS && Gdx.app != null) {
            Gdx.app.log("NIVEL", "NO se encontro el objeto '" + tipo + "' (revisa name/type en Tiled)");
        }
        return null;
    }

    /** Todos los objetos con ese nombre/type/class, en unidades del mundo. */
    public Array<Rectangle> buscarObjetos(String tipo) {
        Array<Rectangle> resultado = new Array<>();
        for (MapObject obj : objetos) {
            if (!coincide(obj, tipo)) continue;
            Rectangle r = rectanguloDe(obj);
            if (r != null) resultado.add(r);
        }
        return resultado;
    }

    public void dispose() {
        mapa.dispose();
    }

    // ---------------------------------------------------------------- internos

    /** Compara con name, type o class (Tiled 1.9+), ignorando mayusculas y espacios. */
    private static boolean coincide(MapObject obj, String tipo) {
        MapProperties p = obj.getProperties();
        return igual(tipo, obj.getName()) || igual(tipo, p.get("type")) || igual(tipo, p.get("class"));
    }

    private static boolean igual(String tipo, Object valor) {
        return valor != null && tipo.equalsIgnoreCase(valor.toString().trim());
    }

    /** Convierte el objeto de Tiled a un rectangulo en unidades del mundo. */
    private static Rectangle rectanguloDe(MapObject obj) {
        if (obj instanceof RectangleMapObject) {
            Rectangle r = ((RectangleMapObject) obj).getRectangle();
            return new Rectangle(r.x * ESCALA, r.y * ESCALA, r.width * ESCALA, r.height * ESCALA);
        }
        // Objetos puestos con "Insertar tile"
        if (obj instanceof TiledMapTileMapObject) {
            MapProperties p = obj.getProperties();
            return new Rectangle(
                numero(p, "x") * ESCALA,
                numero(p, "y") * ESCALA,
                numero(p, "width") * ESCALA,
                numero(p, "height") * ESCALA
            );
        }
        return null;
    }

    private static float numero(MapProperties p, String clave) {
        Object v = p.get(clave);
        return v instanceof Number ? ((Number) v).floatValue() : 0f;
    }

    private void logObjetos() {
        Gdx.app.log("NIVEL", "Mundo: " + anchoPx + " x " + altoPx + " (escala " + ESCALA + "), "
            + objetos.size + " objetos");
        for (MapObject obj : objetos) {
            MapProperties p = obj.getProperties();
            Gdx.app.log("NIVEL", obj.getClass().getSimpleName()
                + " name=" + obj.getName()
                + " type=" + p.get("type")
                + " x=" + p.get("x") + " y=" + p.get("y")
                + " w=" + p.get("width") + " h=" + p.get("height"));
        }
    }

import com.badlogic.gdx.math.Rectangle;
import com.capuccinolovesballerina.game.Utilidades.Constantes;

public class BloqueEspuma {

    private final Rectangle zona;
    private final float vidaInicial;
    private float vida;

    public BloqueEspuma(float x, float y, float ancho, float alto, float vida) {
        this.zona = new Rectangle(x, y, ancho, alto);
        this.vidaInicial = vida;
        this.vida = vida;
    }

    public void actualizar(float delta) {
        vida -= delta;
    }

    public boolean estaViva() {
        return vida > 0f;
    }

    public float getVidaRestante() { return vida; }

    /** El cliente muestra el reloj de la espuma que decide el servidor. */
    public void aplicarVidaRemota(float restante) { vida = Math.max(0f, Math.min(vidaInicial, restante)); }

    public Rectangle getZona() {
        return zona;
    }

    // El reloj pertenece al bloque: cada espuma nueva empieza en el primer frame.
    public int getIndiceFrame() {
        float transcurrido = Math.max(0f, vidaInicial - vida);
        if (vida <= Constantes.DURACION_DISOLUCION_ESPUMA) {
            float disolucion = Constantes.DURACION_DISOLUCION_ESPUMA - vida;
            return 6 + Math.min(1, (int) (disolucion / (Constantes.DURACION_DISOLUCION_ESPUMA / 2f)));
        }
        if (transcurrido < Constantes.DURACION_FORMACION_ESPUMA) {
            return Math.min(3, (int) (transcurrido / (Constantes.DURACION_FORMACION_ESPUMA / 4f)));
        }
        float activo = transcurrido - Constantes.DURACION_FORMACION_ESPUMA;
        return 4 + (int) (activo / Constantes.DURACION_FRAME_ACTIVO_ESPUMA) % 2;
    }

    //alpha es la opacidad, cosa de que se vaya desvaneciendo poco a poco. queda prime confien
    public float getAlpha() {
        float fade = Constantes.DURACION_DISOLUCION_ESPUMA;
        if (vida >= fade) return 1f;
        return Math.max(0f, vida / fade);
    }
}

        cargarPeligros(nivel);

        Rectangle spawn = nivel.buscarObjeto("spawn_cappuccino");
        if (spawn != null) {
            spawnX = spawn.x + (spawn.getWidth() - Constantes.ANCHO_PERSONAJE) / 2f;
            spawnY = spawn.y;
        } else {
            spawnX = 320;
            spawnY = 192;
        }

        Rectangle spawnBallerina = nivel.buscarObjeto("spawn_ballerina");
        if (spawnBallerina != null) {
            spawnBallerinaX = spawnBallerina.x + (spawnBallerina.getWidth() - Constantes.ANCHO_PERSONAJE) / 2f;
            spawnBallerinaY = spawnBallerina.y;
        } else {
            spawnBallerinaX = spawnX + 80f;
            spawnBallerinaY = spawnY;
        }
    }

    private void cargarCortables(Nivel nivel) {
        String[] tipos = {"cuerda", "liana", "barril"};
        for (String tipo : tipos) {
            for (Rectangle zona : nivel.buscarObjetos(tipo)) {
                cortables.add(new ObjetoCortable(tipo, zona));
            }
        }
    }

    private void cargarPeligros(Nivel nivel) {
        for (Rectangle zona : nivel.buscarObjetos("peligro")) {
            peligros.add(zona);                       // lo que se dibuja (no cambia)

            float altoSprite = zona.width / 2f;       // igual que en RenderizadorJuego

                zona.x + Constantes.MARGEN_LATERAL_PELIGRO,
                zona.y,
                zona.width - 2 * Constantes.MARGEN_LATERAL_PELIGRO,
                altoSprite - Constantes.MARGEN_SUPERIOR_PELIGRO
            ));
        }
    }

        labelTiempo.setFontScale(1.5f);
        labelTiempo.setAlignment(Align.center);

        Stack pila = new Stack();
        float anchoMarco = 250f;
        float altoMarco = 80f;
        pila.setSize(anchoMarco, altoMarco);
        pila.add(imagenMarco);
        pila.add(labelTiempo);

        pila.setPosition(
            (Constantes.ANCHO_MUNDO - anchoMarco) / 2f,
            Constantes.ALTO_MUNDO - altoMarco
        );

        stage.addActor(pila);

    }

    public void actualizar(float delta) {
        tiempo += delta;
        int segundo = (int) tiempo;
        if (segundo != ultimoSegundo) {
            ultimoSegundo = segundo;
            labelTiempo.setText(formatear(segundo));
        }
    }

    public void setTiempoRemoto(float segundos) {
        tiempo = segundos;
        actualizar(0f);
    }

        int minutos = totalSegundos / 60;
        int segundos = totalSegundos % 60;
        return String.format("%02d:%02d", minutos, segundos);
    }

    public void dibujar() {
        stage.draw();
    }

    public void reiniciar() {
        tiempo = 0f;
        ultimoSegundo = -1;
        labelTiempo.setText("00:00");
    }

    public void resize(int width, int height) {
        viewport.update(width, height, true);
    }

    public void dispose() {
        stage.dispose();
    }

        if (interfazRed != null) { interfazRed.act(delta); interfazRed.draw(); }

    private void actualizarRed(float delta) {
        EstadoPartida foto = clienteRed.getEstadoPartida();
        if (clienteRed.termino()) {
            estadoRed.setText(clienteRed.getEstadoConexion());
            volverRed.setVisible(true);
            controladorEntrada.soltarTodo();
            Gdx.input.setInputProcessor(interfazRed);
            return;
        }
        estadoRed.setText((clienteRed.getMiRol() == 0 ? "Cappuccino" : "Ballerina") + " - "
            + (foto != null && foto.fase().equals(ProtocolosRed.CUENTA) ? "Inicio en " + foto.cuenta() : clienteRed.getIpServer()));
        if (foto != null && foto.secuencia() > ultimaFoto) {
            if (foto.cortados().length() != elementos.getCortables().size) {
                clienteRed.cerrar("El mapa no coincide con el servidor. Usen la misma version.");
                return;
            }
            ultimaFoto = foto.secuencia();
            if (rondaRed != foto.ronda()) {
                rondaRed = foto.ronda();
                controladorEntrada.soltarTodo();
                menuGameOver.ocultar(); menuVictoria.ocultar(); menuPausa.ocultar();
                animacionesCappuccino.reiniciar(); animacionesBallerina.reiniciar();
            }
            Jugador capRed = foto.cappuccino(), balRed = foto.ballerina();
            cappuccino.setPosicionRemota(capRed.x(), capRed.y(), capRed.mirando(), capRed.suelo());
            ballerina.setPosicionRemota(balRed.x(), balRed.y(), balRed.mirando(), balRed.suelo());
            if (habilidadCap != capRed.habilidad()) {
                habilidadCap = capRed.habilidad();
                animacionesCappuccino.reiniciar();
                animacionesCappuccino.setEstado(com.capuccinolovesballerina.game.graficos.Estado.ABILITY);
            }
            if (habilidadBal != balRed.habilidad()) {
                habilidadBal = balRed.habilidad();
                animacionesBallerina.reiniciar();
                animacionesBallerina.setEstado(com.capuccinolovesballerina.game.graficos.Estado.ABILITY);
            }
            animacionesCappuccino.setEstado(capRed.animacion());
            animacionesBallerina.setEstado(balRed.animacion());
            aplicarPalanca(elementos.getPalancaCappuccino(), foto.palancaCap());
            aplicarPalanca(elementos.getPalancaBallerina(), foto.palancaBal());
            if (elementos.getPuerta() != null) elementos.getPuerta().setAbierta(foto.puerta());
            for (int i = 0; i < elementos.getCortables().size; i++)
                elementos.getCortables().get(i).setCortado(foto.cortados().charAt(i) == '1');
            if (foto.espumaId() == 0 || foto.espumaVida() <= 0f) ballerina.reiniciarHabilidad();
            else {
                if (espumaRed != foto.espumaId() || ballerina.getEspuma() == null)
                    ballerina.setEspumaRemota(foto.espumaX(), foto.espumaY());
                ballerina.getEspuma().aplicarVidaRemota(foto.espumaVida());
            }
            espumaRed = foto.espumaId();
            hud.setTiempoRemoto(foto.tiempo());
            if (foto.fase().equals(ProtocolosRed.DERROTA) && !menuGameOver.estaVisible()) {
                menuPausa.ocultar(); menuGameOver.mostrar();
            }
            if (foto.fase().equals(ProtocolosRed.VICTORIA) && !menuVictoria.estaVisible()) {
                menuPausa.ocultar(); menuVictoria.mostrar();
            }
        }
        if (controladorEntrada.isEscapePresionado() && !menuGameOver.estaVisible() && !menuVictoria.estaVisible()) {
            if (menuPausa.estaVisible()) menuPausa.ocultar();
            else { menuPausa.mostrar(); controladorEntrada.soltarTodo(); }
        }
        boolean puedeMover = foto != null && foto.fase().equals(ProtocolosRed.JUGANDO) && !menuPausa.estaVisible();
        if (menuPausa.estaVisible()) menuPausa.actualizar(delta);
        else if (menuGameOver.estaVisible()) menuGameOver.actualizar(delta);
        else if (menuVictoria.estaVisible()) menuVictoria.actualizar(delta);
        else Gdx.input.setInputProcessor(controladorEntrada);
        boolean cap = clienteRed.getMiRol() == 0;
        clienteRed.enviarControles(puedeMover && (cap ? controladorEntrada.isIzquierda() : controladorEntrada.isIzquierdaBallerina()),
            puedeMover && (cap ? controladorEntrada.isDerecha() : controladorEntrada.isDerechaBallerina()),
            puedeMover && (cap ? controladorEntrada.isSaltoPresionado() : controladorEntrada.isSaltoPresionadoBallerina()),
            puedeMover && (cap ? controladorEntrada.isHabilidadPresionada() : controladorEntrada.isHabilidadPresionadaBallerina()),
            puedeMover && (cap ? controladorEntrada.isInteractuarPresionado() : controladorEntrada.isInteractuarPresionadoBallerina()));
        controladorEntrada.limpiarEventos();
        animacionesCappuccino.actualizar(delta);
        animacionesBallerina.actualizar(delta);
    }

package com.capuccinolovesballerina.game.red;

import com.capuccinolovesballerina.game.graficos.Estado;
import java.nio.charset.StandardCharsets;

/** Protocolo de texto UTF-8. Separador de campos: |. Dentro de un jugador: coma. */
public final class ProtocolosRed {
    private ProtocolosRed() {}
    public static final int PUERTO = 9992, MAX_PAQUETE = 1400;
    public static final long TIMEOUT_MS = 5000, BUSQUEDA_MS = 10000;
    public static final String CONEXION = "CONEXION", OK = "OK", ASIGNADO = "ASIGNADO";
    public static final String REGISTRO = "REGISTRO", LISTO = "LISTO", PONG = "PONG";
    public static final String LLENO = "LLENO", CONTROLES = "CONTROLES", ESTADO = "ESTADO";
    public static final String PING = "PING", SALIR = "SALIR", DESCONECTADO = "DESCONECTADO", REINICIO = "REINICIO";
    public static final String ESPERA = "ESPERA", CUENTA = "CUENTA", JUGANDO = "JUGANDO", DERROTA = "DERROTA", VICTORIA = "VICTORIA";
    public static final String MAPA = "mapas/NIVEL 0.tmx";
    public static byte[] bytes(String mensaje) { return mensaje.getBytes(StandardCharsets.UTF_8); }
    public static String[] partes(String mensaje) { return mensaje.split("\\|", -1); }
    public static int bit(boolean valor) { return valor ? 1 : 0; }
    public static boolean booleano(String valor) {
        if (!valor.equals("0") && !valor.equals("1")) throw new IllegalArgumentException("Booleano invalido");
        return valor.equals("1");
    }
    public static long natural(String valor) {
        long numero = Long.parseLong(valor);
        if (numero < 0) throw new IllegalArgumentException("Numero negativo");
        return numero;
    }
    public static float numero(String valor) {
        float numero = Float.parseFloat(valor);
        if (!Float.isFinite(numero) || Math.abs(numero) > 1_000_000f) throw new IllegalArgumentException("Numero invalido");
        return numero;
    }
    // Los contadores acumulados repiten los eventos perdidos y evitan ejecutarlos dos veces.
    public record Controles(boolean izquierda, boolean derecha, long salto, long habilidad, long interactuar) {
        public static final Controles VACIOS = new Controles(false, false, 0, 0, 0);
        public String mensaje(String sesion, long ronda, long secuencia) {
            return CONTROLES + "|" + sesion + "|" + ronda + "|" + secuencia + "|" + bit(izquierda)
                + "|" + bit(derecha) + "|" + salto + "|" + habilidad + "|" + interactuar;
        }
        public static Controles leer(String[] p) {
            if (p.length != 9) throw new IllegalArgumentException("Controles incompletos");
            return new Controles(booleano(p[4]), booleano(p[5]), natural(p[6]), natural(p[7]), natural(p[8]));
        }
    }
    public record Jugador(float x, float y, boolean mirando, boolean suelo, Estado animacion, long habilidad) {
        public String texto() { return x + "," + y + "," + bit(mirando) + "," + bit(suelo) + "," + animacion + "," + habilidad; }
        public static Jugador leer(String texto) {
            String[] p = texto.split(",", -1);
            if (p.length != 6) throw new IllegalArgumentException("Jugador incompleto");
            return new Jugador(numero(p[0]), numero(p[1]), booleano(p[2]), booleano(p[3]), Estado.valueOf(p[4]), natural(p[5]));
        }
    }
    /** Foto completa e inmutable: pasa del hilo de red al grafico. */
    public record EstadoPartida(long secuencia, long ronda, String fase, int cuenta, float tiempo,
        Jugador cappuccino, Jugador ballerina, boolean palancaCap, boolean palancaBal, boolean puerta,
        String cortados, long espumaId, float espumaX, float espumaY, float espumaVida) {
        public String mensaje(String sesion) {
            return ESTADO + "|" + sesion + "|" + secuencia + "|" + ronda + "|" + fase + "|" + cuenta + "|" + tiempo
                + "|" + cappuccino.texto() + "|" + ballerina.texto() + "|" + bit(palancaCap) + "|" + bit(palancaBal)
                + "|" + bit(puerta) + "|" + cortados + "|" + espumaId + "|" + espumaX + "|" + espumaY + "|" + espumaVida;
        }
        public static EstadoPartida leer(String[] p) {
            if (p.length != 17) throw new IllegalArgumentException("Estado incompleto");
            if (!java.util.Set.of(ESPERA, CUENTA, JUGANDO, DERROTA, VICTORIA).contains(p[4])) throw new IllegalArgumentException("Fase invalida");
            int cuenta = Integer.parseInt(p[5]);
            float tiempo = numero(p[6]), vida = numero(p[16]);
            if (cuenta < 0 || cuenta > 5 || tiempo < 0 || vida < 0 || vida > 4f || !p[12].matches("[01]*"))
                throw new IllegalArgumentException("Estado invalido");
            return new EstadoPartida(natural(p[2]), natural(p[3]), p[4], cuenta, tiempo,
                Jugador.leer(p[7]), Jugador.leer(p[8]), booleano(p[9]), booleano(p[10]), booleano(p[11]),
                p[12], natural(p[13]), numero(p[14]), numero(p[15]), vida);
        }
    }
}

    public static void main(String[] args) throws Exception {

    public void ejecutar() {
        red.start();
        System.out.println("[RED] Servidor UDP en 0.0.0.0:" + PUERTO + ". Esperando dos jugadores.");
        System.out.println("[RED] Usar la IP Ethernet de esta PC para conexion manual. Ctrl+C para cerrar.");
        mostrarDirecciones();

            while (leidos++ < 512 && (recibido = red.siguiente()) != null) procesar(recibido);

            for (Conexion cliente : clientes) {

                    break;
                }

                if (ahora >= finCuenta) { fase = JUGANDO; System.out.println("[RED] INICIO de ronda " + ronda); }
            }

    private void procesar(HiloServidor.Recibido recibido) {
        String[] p = partes(recibido.mensaje());
        InetSocketAddress origen = recibido.origen();
        try {
            if (p.length == 1 && p[0].equals(CONEXION)) {
                // Descubrir no ocupa un cupo: el mismo equipo puede emitir por varias interfaces.
                red.enviar(origen, OK);
                System.out.println("[RED] CONEXION desde " + origen + ": OK enviado, esperando REGISTRO directo.");
                return;
            }
            if (p.length == 1 && p[0].equals(REGISTRO)) { conectar(origen); return; }
            int rol = buscar(origen);
            if (rol < 0 || p.length < 2 || !clientes[rol].sesion.equals(p[1])) return;
            Conexion c = clientes[rol];
            if (p[0].equals(LISTO) && p.length == 2) {
                if (!c.listo) System.out.println("[RED] " + origen + " confirmo su personaje (LISTO).");
                c.listo = true;
                if (fase.equals(ESPERA) && clientes[0] != null && clientes[1] != null
                    && clientes[0].listo && clientes[1].listo) comenzarCuenta();
            } else if (p[0].equals(CONTROLES)) {
                Controles controles = Controles.leer(p);
                long rondaRecibida = natural(p[2]), numero = natural(p[3]);
                if (numero <= c.ultimaSecuencia || rondaRecibida != ronda) return;
                if (controles.salto() < c.controles.salto() || controles.habilidad() < c.controles.habilidad()
                    || controles.interactuar() < c.controles.interactuar()) return;
                if (fase.equals(JUGANDO)) {
                    entrada.aplicarControlesRed(rol, controles.izquierda(), controles.derecha(),
                        controles.salto() > c.controles.salto(), controles.habilidad() > c.controles.habilidad(),
                        controles.interactuar() > c.controles.interactuar());
                }
                c.controles = controles;
                c.ultimaSecuencia = numero;
                c.ultimoControl = System.nanoTime();
            } else if (p[0].equals(REINICIO) && p.length == 4) {
                long rondaRecibida = natural(p[2]), numero = natural(p[3]);
                if (rondaRecibida != ronda || numero <= c.reinicio) return;
                c.reinicio = numero;
                if (fase.equals(DERROTA) || fase.equals(VICTORIA)) comenzarCuenta();
            } else if (p[0].equals(SALIR) && p.length == 2) {
                desconectar("El otro jugador salio de la partida"); return;

                // Equivale a ESTAS_AHI / ESTOY_ACA en SpeedG2L.
                red.enviar(origen, PONG + "|" + c.sesion);
            } else return;
            c.ultimoMensaje = System.nanoTime();

        }

    private void mostrarDirecciones() {
        try {
            var interfaces = NetworkInterface.getNetworkInterfaces();

    public static List<InetSocketAddress> destinosDescubrimiento() throws SocketException, UnknownHostException {

package com.capuccinolovesballerina.game.red.hilos;

import com.capuccinolovesballerina.game.red.ProtocolosRed;
import java.io.IOException;
import java.net.*;
import java.nio.charset.StandardCharsets;
import java.util.concurrent.ArrayBlockingQueue;

/** Solo recibe datagramas. Servidor modifica la partida en su propio hilo. */
public class HiloServidor extends Thread {
    public record Recibido(InetSocketAddress origen, String mensaje) {}
    private final DatagramSocket socket;
    private final ArrayBlockingQueue<Recibido> bandeja = new ArrayBlockingQueue<>(512);
    private volatile boolean fin;
    public HiloServidor() throws SocketException {
        super("UDP-servidor");
        socket = new DatagramSocket(ProtocolosRed.PUERTO);
    }
    @Override public void run() {
        while (!fin) {
            byte[] buffer = new byte[ProtocolosRed.MAX_PAQUETE + 1];
            DatagramPacket paquete = new DatagramPacket(buffer, buffer.length);
            try {
                socket.receive(paquete);
                if (paquete.getLength() > ProtocolosRed.MAX_PAQUETE) continue;
                bandeja.offer(new Recibido((InetSocketAddress) paquete.getSocketAddress(),
                    new String(paquete.getData(), 0, paquete.getLength(), StandardCharsets.UTF_8)));
            } catch (IOException error) {
                if (!fin) System.err.println("[RED] Error de recepcion: " + error.getMessage());
            }
        }
    }
    public Recibido siguiente() { return bandeja.poll(); }
    public void enviar(InetSocketAddress destino, String mensaje) {
        byte[] datos = ProtocolosRed.bytes(mensaje);
        if (datos.length > ProtocolosRed.MAX_PAQUETE) throw new IllegalStateException("Estado demasiado grande para un datagrama");
        try { socket.send(new DatagramPacket(datos, datos.length, destino)); }
        catch (IOException error) { if (!fin) System.err.println("[RED] Error de envio: " + error.getMessage()); }
    }
    public void detener() { fin = true; socket.close(); interrupt(); }
}

            izquierda = true;
        }
        if (keycode == Input.Keys.D) {
            derecha = true;
        }

        // Salto Cappuccino
        if (keycode == Input.Keys.W) {
            if (!saltoW && !saltoEspacio) saltoPresionado = true;
            saltoW = true;
        }
        if (keycode == Input.Keys.SPACE) {
            if (!saltoW && !saltoEspacio) saltoPresionado = true;
            saltoEspacio = true;
        }

        if (keycode == Input.Keys.E) {
            interactuarPresionado = true;
        }
        if (keycode == Input.Keys.F) {
            habilidadPresionada = true;
        }

        if (keycode == Input.Keys.LEFT) {
            izquierdaBallerina = true;
        }
        if (keycode == Input.Keys.RIGHT) {
            derechaBallerina = true;
        }

        if (keycode == Input.Keys.UP) {
            saltoPresionadoBallerina = true;
        }

        if (keycode == Input.Keys.SHIFT_RIGHT) {
            habilidadPresionadaBallerina = true;
        }

        if (keycode == Input.Keys.ENTER) {
            interactuarPresionadoBallerina = true;
        }

        return true;
    }

        if (keycode == Input.Keys.A) {
            izquierda = false;

        if (keycode == Input.Keys.D) {

        return true;

        MenuVictoria menuVictoria,
        HUD hud,
        ControladorEntradaJugador entrada,
        CapuccinoAssassino cappuccino,
        BallerinaCapuccina ballerina,
        AnimacionesCappuccino animacionesCappuccino,
        AnimacionesBallerina animacionesBallerina
    ) {
        this.nivel = nivel;
        this.elementos = elementos;
        this.menuPausa = menuPausa;
        this.menuGameOver = menuGameOver;
        this.menuVictoria = menuVictoria;
        this.hud = hud;
        this.entrada = entrada;
        this.cappuccino = cappuccino;
        this.ballerina = ballerina;
        this.animacionesCappuccino = animacionesCappuccino;
        this.animacionesBallerina = animacionesBallerina;
    }

    public void actualizar(float delta) {

        if (menuPausa != null && entrada.isEscapePresionado()) {
            if (menuPausa.estaVisible()) {
                menuPausa.ocultar();
            } else if (!menuGameOver.estaVisible() && !menuVictoria.estaVisible()) {
                menuPausa.mostrar();
                entrada.soltarTodo();
            }
        }

        if (menuPausa != null && menuPausa.estaVisible()) {
            menuPausa.actualizar(delta);
            return;
        }

        if (derrota) {
            if (menuGameOver != null) menuGameOver.actualizar(delta);
            return;
        }

        if (victoria) {
            if (menuVictoria != null) menuVictoria.actualizar(delta);
            return;
        }

        if (menuPausa != null && Gdx.input != null && Gdx.input.getInputProcessor() != entrada) {
            Gdx.input.setInputProcessor(entrada);
        }

        if (hud != null) hud.actualizar(delta);

        ballerina.actualizarHabilidad(delta);

        if (entrada.isHabilidadPresionadaBallerina()) {
            ballerina.crearEspuma();
        }


        Array<Rectangle> solidos = new Array<>(nivel.getColisiones().getSolidos());

        for (ObjetoCortable cortable : elementos.getCortables()) {
            if (cortable.bloquea()) {
                solidos.add(cortable.getZona());
            }
        }

        BloqueEspuma espuma = ballerina.getEspuma();
        if (espuma != null) {
            solidos.add(espuma.getZona());
        }

        cappuccino.actualizar(
            delta,
            entrada.isIzquierda(),           // A
            entrada.isDerecha(),             // D
            entrada.isSaltoPresionado(),     // W / Espacio
            solidos
        );

        ballerina.actualizar(
            delta,
            entrada.isIzquierdaBallerina(),      // ←
            entrada.isDerechaBallerina(),        // →
            entrada.isSaltoPresionadoBallerina(),// ↑
            solidos
        );


        if (entrada.isHabilidadPresionada()) {
            estadoCappuccino(Estado.ABILITY);
            cortar();
        } else if (!cappuccino.isEnSuelo()) {
            estadoCappuccino(Estado.JUMP);
        } else if (entrada.isIzquierda() || entrada.isDerecha()) {
            estadoCappuccino(Estado.WALK);
        } else {
            estadoCappuccino(Estado.IDLE);
        }

        if (entrada.isHabilidadPresionadaBallerina()) {
            estadoBallerina(Estado.ABILITY);
        } else if (!ballerina.isEnSuelo()) {
            estadoBallerina(Estado.JUMP);
        } else if (entrada.isIzquierdaBallerina() || entrada.isDerechaBallerina()) {
            estadoBallerina(Estado.WALK);
        } else {
            estadoBallerina(Estado.IDLE);
        }

        if (animacionesCappuccino != null) animacionesCappuccino.actualizar(delta);
        if (animacionesBallerina != null) animacionesBallerina.actualizar(delta);


        if (estaMuerto(cappuccino) || estaMuerto(ballerina)) {
            mostrarGameOver();
            return;
        }


        comprobarVictoria();

        if (victoria) {
            return;
        }
        actualizarPalancas();
        entrada.limpiarEventos();
    }

    private void actualizarPalancas() {
        Palanca palCap = elementos.getPalancaCappuccino();
        Palanca palBal = elementos.getPalancaBallerina();

        if (entrada.isInteractuarPresionado()
            && estaCerca(cappuccino, palCap)) {
            intentarAlternar(palCap);
        }

        if (entrada.isInteractuarPresionadoBallerina()
            && estaCerca(ballerina, palBal)) {
            intentarAlternar(palBal);
        }
    }
    private boolean estaCerca(Player personaje, Palanca palanca) {
        if (palanca == null) return false;
        Rectangle c = personaje.getCaja();
        Rectangle alcance = new Rectangle(
            c.x - ALCANCE_PALANCA, c.y - ALCANCE_PALANCA,
            c.width + 2 * ALCANCE_PALANCA, c.height + 2 * ALCANCE_PALANCA);
        return alcance.overlaps(palanca.getZona());
    }
    private void intentarAlternar(Palanca palanca) {
        Puerta puerta = elementos.getPuerta();

        // No dejar cerrar la puerta con alguien parado adentro
        boolean seCerraria = palanca.estaActivada() && puerta != null && puerta.estaAbierta();
        if (seCerraria && alguienEnLaPuerta()) return;

        palanca.alternar();
        actualizarPuerta();
    }


    private void actualizarPuerta() {
        Puerta puerta = elementos.getPuerta();
        Palanca palCap = elementos.getPalancaCappuccino();
        Palanca palBal = elementos.getPalancaBallerina();
        if (puerta == null || (palCap == null && palBal == null)) return;

        // si un nivel solo tiene una palanca, la que falta no bloquea
        boolean capOk = palCap == null || palCap.estaActivada();
        boolean balOk = palBal == null || palBal.estaActivada();
        puerta.setAbierta(capOk && balOk);
    }
    private boolean alguienEnLaPuerta() {
        Puerta puerta = elementos.getPuerta();
        return puerta != null
            && (cappuccino.getCaja().overlaps(puerta.getZona())
            || ballerina.getCaja().overlaps(puerta.getZona()));
    }


    private void comprobarVictoria() {
        Puerta puerta = elementos.getPuerta();

        if (puerta == null || !puerta.estaAbierta()) {
            return;
        }

        Rectangle zona = puerta.getZona();

        if (cappuccino.getCaja().overlaps(zona) && ballerina.getCaja().overlaps(zona)) {
            entrada.soltarTodo();
            victoria = true;
            if (menuVictoria != null) menuVictoria.mostrar();
        }
    }

    private boolean estaMuerto(Player personaje) {
        if (personaje.getY() < -personaje.getAlto()) {
            return true;
        }

        for (Rectangle peligro : elementos.getHitboxPeligros()) {
            if (personaje.getCaja().overlaps(peligro)) {
                return true;
            }
        }

        return false;
    }

    private void mostrarGameOver() {
        entrada.soltarTodo();
        derrota = true;
        if (menuGameOver != null) menuGameOver.mostrar();
    }

    private void cortar() {
        Rectangle caja = cappuccino.getCaja();
        float alcance = 40f;
        Rectangle zonaCorte;

        if (cappuccino.isMirandoDerecha()) {
            zonaCorte = new Rectangle(caja.x + caja.width, caja.y, alcance, caja.height);
        } else {
            zonaCorte = new Rectangle(caja.x - alcance, caja.y, alcance, caja.height);
        }

        for (ObjetoCortable cortable : elementos.getCortables()) {
            if (!cortable.estaCortado() && zonaCorte.overlaps(cortable.getZona())) {
                cortable.setCortado(true);
            }
        }
    }

    public void reiniciar() {
        derrota = false;
        victoria = false;
        estados[0] = estados[1] = Estado.IDLE;
        cappuccino.reaparecer(elementos.getSpawnX(), elementos.getSpawnY());
        ballerina.reaparecer(elementos.getSpawnBallerinaX(), elementos.getSpawnBallerinaY());

        ballerina.reiniciarHabilidad();

        Puerta puerta = elementos.getPuerta();

        if (puerta != null) {
            puerta.setAbierta(false);
        }

        Palanca palCap = elementos.getPalancaCappuccino();
        Palanca palBal = elementos.getPalancaBallerina();
        if (palCap != null && palCap.estaActivada()) palCap.alternar();
        if (palBal != null && palBal.estaActivada()) palBal.alternar();

        for (ObjetoCortable cortable : elementos.getCortables()) {
            cortable.setCortado(false);
        }

        entrada.soltarTodo();
        if (animacionesCappuccino != null) animacionesCappuccino.reiniciar();
        if (animacionesBallerina != null) animacionesBallerina.reiniciar();
        if (hud != null) hud.reiniciar();
    }
}

import java.util.InputMismatchException;
import java.util.Scanner;

public class Entrada {

    private Scanner s = new Scanner(System.in);

    public int ingresarEntero(int min, int max) {
        boolean error;
        int numero = 0;

        do {
            error = false;

            try {
                numero = s.nextInt();

                if (numero > max || numero < min) {
                    System.out.println(
                        "Ingrese un número entre " + min + " y " + max
                    );
                    error = true;
                }

            } catch (InputMismatchException e) {
                System.out.println("Ingrese un valor numérico");
                error = true;

            } finally {
                s.nextLine();
            }

        } while (error);

        return numero;
    }

    public String ingresarTexto() {
        return s.nextLine();
    }
}

    private final OrthographicCamera camera;
    private final OrthogonalTiledMapRenderer renderizador;
    private final ShapeRenderer shapeRenderer;
    private final SpriteBatch batch;

    private Texture texturaAbierta;
    private Texture texturaPalancaOff;

    private Texture texturaPinchos;

    private Animation<TextureRegion> animacionLianas;
    private float tiempoLianas = 0f;
    private Texture texturaEspuma;

    public RenderizadorJuego(
        OrthographicCamera camera,
        Nivel nivel,
        CapuccinoAssassino cappuccino,

        this.animacionesCappuccino = animacionesCappuccino;

    private void cargarTexturas() {
        texturaCerrada = new Texture("objetos/puerta_cerrada.png");
        texturaAbierta = new Texture("objetos/puerta_abierta.png");
        texturaPalancaOff = new Texture("objetos/palanca_apagada.png");
        texturaPalancaOn = new Texture("objetos/palanca_encendida.png");
        texturaPinchos = new Texture("objetos/pinchos.png");
        texturaLiana = new Texture("sprites/lianas_animadas.png");
        TextureRegion[] framesLiana = new TextureRegion[CANTIDAD_FRAMES_LIANA];
        for (int i = 0; i < framesLiana.length; i++) {

                i * ANCHO_FRAME_LIANA, 0, ANCHO_FRAME_LIANA, ALTO_FRAME_LIANA);

        animacionLianas.setPlayMode(Animation.PlayMode.LOOP_PINGPONG);

    private void dibujarCortables() {
        shapeRenderer.setProjectionMatrix(camera.combined);
        shapeRenderer.begin(ShapeRenderer.ShapeType.Filled);

        for (ObjetoCortable cortable : elementos.getCortables()) {
            if (cortable.estaCortado()) continue;
            if ("liana".equals(cortable.getTipo())) continue;

            if ("cuerda".equals(cortable.getTipo())) {
                shapeRenderer.setColor(0.8f, 0.7f, 0.4f, 1f);
            } else {
                shapeRenderer.setColor(0.55f, 0.35f, 0.15f, 1f);
            }

            Rectangle z = cortable.getZona();
            shapeRenderer.rect(z.x, z.y, z.width, z.height);
        }

    private void dibujarEspuma() {
        BloqueEspuma espuma = ballerina.getEspuma();

        float colorAnterior = batch.getPackedColor();
        batch.setColor(1f, 1f, 1f, espuma.getAlpha());
        batch.draw(framesEspuma[espuma.getIndiceFrame()], z.x, z.y, z.width, z.height);
        batch.setPackedColor(colorAnterior);

        Texture textura = puerta.estaAbierta() ? texturaAbierta : texturaCerrada;

        float ancho = 96f;
        float alto = 128f;
        batch.draw(textura, z.x + (z.width - ancho) / 2f, z.y, ancho, alto);

        if (palanca == null) return;

        Rectangle z = palanca.getZona();
        Texture textura = palanca.estaActivada() ? texturaPalancaOn : texturaPalancaOff;

        float ancho = Constantes.ANCHO_PALANCA;   // antes: 48f
        float alto = Constantes.ALTO_PALANCA;     // antes: 56f
        batch.setColor(tinte);
        batch.draw(textura, z.x + (z.width - ancho) / 2f, z.y, ancho, alto);
        batch.setColor(Color.WHITE);
    }

    private void dibujarPeligros() {
        for (Rectangle peligro : elementos.getPeligros()) {
            batch.draw(texturaPinchos, peligro.x, peligro.y, peligro.width, peligro.width / 2f);
        }
    }


    private void dibujarLianas() {
        TextureRegion frame = animacionLianas.getKeyFrame(tiempoLianas);

        for (ObjetoCortable cortable : elementos.getCortables()) {
            if (cortable.estaCortado()) continue;
            if (!"liana".equals(cortable.getTipo())) continue;

            Rectangle z = cortable.getZona();

            // Tamaño visual independiente de la hitbox
            float altoVisual = z.height;
            float anchoVisual = z.width * 2.5f;

            // Centrar la imagen respecto a la hitbox
            float xVisual = z.x + (z.width - anchoVisual) / 2f;

            batch.draw(frame, xVisual, z.y, anchoVisual, altoVisual);
        }
    }


    private void dibujarCappuccino() {
        TextureRegion frame = animacionesCappuccino.getFrameActual();

        float tam = Constantes.TAM_SPRITE_PERSONAJE;
        float x = cappuccino.getX() - (tam - cappuccino.getAncho()) / 2f;
        float y = cappuccino.getY();

        if (cappuccino.isMirandoDerecha()) {
            batch.draw(frame, x, y, tam, tam);
        } else {
            batch.draw(frame, x + tam, y, -tam, tam);
        }
    }

    private void dibujarBallerina() {
        TextureRegion frame = animacionesBallerina.getFrameActual();

        float tam = Constantes.TAM_SPRITE_PERSONAJE;
        float x = ballerina.getX() - (tam - ballerina.getAncho()) / 2f;
        float y = ballerina.getY();

        if (ballerina.isMirandoDerecha()) {
            batch.draw(frame, x, y, tam, tam);
        } else {
            batch.draw(frame, x + tam, y, -tam, tam);
        }
    }

    public void dispose() {
        renderizador.dispose();
        shapeRenderer.dispose();
        batch.dispose();
        texturaAbierta.dispose();
        texturaCerrada.dispose();
        texturaPalancaOff.dispose();
        texturaPalancaOn.dispose();
        texturaPinchos.dispose();
        texturaLiana.dispose();
        texturaEspuma.dispose();
    }
}

/**
 * A helper object for game startup, featuring three utilities related to LWJGL3 on various operating systems.
 * <p>
 * The utilities are as follows:
 * <ul>
 *  <li> Windows: Prevents a common crash related to LWJGL3's extraction of shared library files.</li>
 *  <li> macOS: Spawns a child JVM process with {@code -XstartOnFirstThread} in the JVM args (if it was not already).
 *  This is required for LWJGL3 to work on macOS.</li>
 *  <li> Linux (NVIDIA GPUs only): Spawns a child JVM process with the {@code __GL_THREADED_OPTIMIZATIONS}
 *  {@link System#getenv(String) Environment Variable} set to {@code 0} (if it was not already). This is required for
 *  LWJGL3 to work on Linux with NVIDIA GPUs.</li>
 * </ul>
 * <a href="https://jvm-gaming.org/t/starting-jvm-on-mac-with-xstartonfirstthread-programmatically/57547">Based on this java-gaming.org post by kappa</a>
 * @author damios
 */
public class StartupHelper {

	private StartupHelper() {}

	private static final String JVM_RESTARTED_ARG = "jvmIsRestarted";

	/**
	 * Must only be called on Linux. Check OS first (or use short-circuit evaluation)!
	 * @return whether NVIDIA drivers are present on Linux.
	 */
	public static boolean isLinuxNvidia() {
		String[] drivers = new File("/proc/driver").list(
			(dir, path) -> path.toUpperCase(Locale.ROOT).contains("NVIDIA")
		);
		if (drivers == null) return false;
		return drivers.length > 0;
	}

	/**
	 * Applies the utilities as described by {@link StartupHelper}'s Javadoc.
	 * <p>
	 * All {@link System#getenv() Environment Variables} are copied to the child JVM process (if it is spawned), as
	 * specified by {@link ProcessBuilder#environment()}; the same applies for
	 * {@link System#getProperties() System Properties}.
	 * <p>
	 * <b>Usage:</b>
	 * <pre><code>
	 * public static void main(String[] args) {
	 * 	 if (StartupHelper.startNewJvmIfRequired()) return;
	 * 	 // ... The rest of main() goes here, as normal.
	 * }
	 * </code></pre>
	 * @return whether a child JVM process was spawned or not.
	 */
	public static boolean startNewJvmIfRequired() {
		return startNewJvmIfRequired(true);
	}

	/**
	 * Applies the utilities as described by {@link StartupHelper}'s Javadoc.
	 * <p>
	 * All {@link System#getenv() Environment Variables} are copied to the child JVM process (if it is spawned), as
	 * specified by {@link ProcessBuilder#environment()}; the same applies for
	 * {@link System#getProperties() System Properties}.
	 * <p>
	 * <b>Usage:</b>
	 * <pre><code>
	 * public static void main(String[] args) {
	 *   // The parameter on the next line could instead be false if you don't want to inherit IO.
	 * 	 if (StartupHelper.startNewJvmIfRequired(true)) return;
	 * 	 // ... The rest of main() goes here, as normal.
	 * }
	 * </code></pre>
	 * @param inheritIO whether I/O should be inherited in the child JVM process. Please note that enabling this will
	 *                  block the thread until the child JVM process stops executing.
	 * @return whether a child JVM process was spawned or not.
	 */
	public static boolean startNewJvmIfRequired(boolean inheritIO) {
		String osName = System.getProperty("os.name").toLowerCase(Locale.ROOT);
		if (osName.contains("mac")) return startNewJvm0(/*isMac =*/ true, inheritIO);
		if (osName.contains("windows")) {
			// Here, we are trying to work around an issue with how LWJGL3 loads its extracted .dll files.
			// By default, LWJGL3 extracts to the directory specified by "java.io.tmpdir": usually, the user's home.
			// If the user's name has non-ASCII (or some non-alphanumeric) characters in it, that would fail.
			// By extracting to the relevant "ProgramData" folder, which is usually "C:\ProgramData", we avoid this.
			// We also temporarily change the "user.name" property to one without any chars that would be invalid.
			// We revert our changes immediately after loading LWJGL3 natives.
			String programData = System.getenv("ProgramData");
			if (programData == null) programData = "C:\\Temp"; // if ProgramData isn't set, try some fallback.
			String prevTmpDir = System.getProperty("java.io.tmpdir", programData);
			String prevUser = System.getProperty("user.name", "libGDX_User");
			System.setProperty("java.io.tmpdir", programData + "\\libGDX-temp");
			System.setProperty(
				"user.name",
				("User_" + prevUser.hashCode() + "_GDX" + Version.VERSION).replace('.', '_')
			);
			Lwjgl3NativesLoader.load();
			System.setProperty("java.io.tmpdir", prevTmpDir);
			System.setProperty("user.name", prevUser);
			return false;
		}
		return startNewJvm0(/*isMac =*/ false, inheritIO);
	}

	private static final String MAC_JRE_ERR_MSG = "A Java installation could not be found. If you are distributing this app with a bundled JRE, be sure to set the '-XstartOnFirstThread' argument manually!";
	private static final String LINUX_JRE_ERR_MSG = "A Java installation could not be found. If you are distributing this app with a bundled JRE, be sure to set the environment variable '__GL_THREADED_OPTIMIZATIONS' to '0'!";
	private static final String CHILD_LOOP_ERR_MSG = "The current JVM process is a spawned child JVM process, but StartupHelper has attempted to spawn another child JVM process! This is a broken state, and should not normally happen! Your game may crash or not function properly!";

	/**
	 * Spawns a child JVM process if on macOS, or on Linux with NVIDIA drivers.
	 * <p>
	 * All {@link System#getenv() Environment Variables} are copied to the child JVM process (if it is spawned), as
	 * specified by {@link ProcessBuilder#environment()}; the same applies for
	 * {@link System#getProperties() System Properties}.
	 * @param isMac whether the current OS is macOS. If this is `false` then the current OS is assumed to be Linux (and
	 *             an immediate check for NVIDIA drivers is performed).
	 * @param inheritIO whether I/O should be inherited in the child JVM process. Please note that enabling this will
	 *                 block the thread until the child JVM process stops executing.
	 * @return whether a child JVM process was spawned or not.
	 */
	public static boolean startNewJvm0(boolean isMac, boolean inheritIO) {
		long processID = getProcessID(isMac);
		if (!isMac) {
			// No need to restart non-NVIDIA Linux
			if (!isLinuxNvidia()) return false;
			// check whether __GL_THREADED_OPTIMIZATIONS is already disabled
			if ("0".equals(System.getenv("__GL_THREADED_OPTIMIZATIONS"))) return false;
		} else {
			// There is no need for -XstartOnFirstThread on Graal native image
			if (!System.getProperty("org.graalvm.nativeimage.imagecode", "").isEmpty()) return false;

			// Checks if we are already on the main thread, such as from running via Construo.
			long objcMsgSend = ObjCRuntime.getLibrary().getFunctionAddress("objc_msgSend");
			long nsThread = ObjCRuntime.objc_getClass("NSThread");
			long currentThread = JNI.invokePPP(nsThread, ObjCRuntime.sel_getUid("currentThread"), objcMsgSend);
			boolean isMainThread = JNI.invokePPZ(currentThread, ObjCRuntime.sel_getUid("isMainThread"), objcMsgSend);
			if (isMainThread) return false;

			if ("1".equals(System.getenv("JAVA_STARTED_ON_FIRST_THREAD_" + processID))) return false;
		}

		// Check whether this JVM process is a child JVM process already.
		// This state shouldn't usually be reachable, but this stops us from endlessly spawning new child JVM processes.
		if ("true".equals(System.getProperty(JVM_RESTARTED_ARG))) {
			System.err.println(CHILD_LOOP_ERR_MSG);
			return false;
		}

		// Spawn the child JVM process with updated environment variables or JVM args
		List<String> jvmArgs = new ArrayList<>();
		// The following line is used assuming you target Java 8, the minimum for LWJGL3.
		String javaExecPath = System.getProperty("java.home") + "/bin/java";
		// If targeting Java 9 or higher, you could use the following instead of the above line:
		//String javaExecPath = ProcessHandle.current().info().command().orElseThrow()
		if (!(new File(javaExecPath).exists())) {
			System.err.println(getJreErrMsg(isMac));
			return false;
		}

		jvmArgs.add(javaExecPath);
		if (isMac) jvmArgs.add("-XstartOnFirstThread");
		jvmArgs.add("-D" + JVM_RESTARTED_ARG + "=true");
		jvmArgs.addAll(ManagementFactory.getRuntimeMXBean().getInputArguments());
		jvmArgs.add("-cp");
		jvmArgs.add(System.getProperty("java.class.path"));
		String mainClass = System.getenv("JAVA_MAIN_CLASS_" + processID);
		if (mainClass == null) {
			StackTraceElement[] trace = Thread.currentThread().getStackTrace();
			if (trace.length > 0) mainClass = trace[trace.length - 1].getClassName();
			else {
				System.err.println("The main class could not be determined.");
				return false;
			}
		}
		jvmArgs.add(mainClass);

		try {
			ProcessBuilder processBuilder = new ProcessBuilder(jvmArgs);
			if (!isMac) processBuilder.environment().put("__GL_THREADED_OPTIMIZATIONS", "0");

			if (!inheritIO) processBuilder.start();
			else processBuilder.inheritIO().start().waitFor();
		} catch (Exception e) {
			System.err.println("There was a problem restarting the JVM.");
			// noinspection CallToPrintStackTrace
			e.printStackTrace();
		}

		return true;
	}

	private static String getJreErrMsg(boolean isMac) {
		if (isMac) return MAC_JRE_ERR_MSG;
		else return LINUX_JRE_ERR_MSG;
	}

	private static long getProcessID(boolean isMac) {
		if (isMac) return LibC.getpid();
		else return UNISTD.getpid();
	}
}

# Capuccino Loves Ballerina

## Integrantes del grupo

- Guillermo Bordoli
- Agustín Bruno
- Santino Fierro

## Descripción del videojuego

Capuccino Loves Ballerina es un videojuego cooperativo de plataformas y puzles en 2D para PC, inspirado en las mecánicas principales de Fireboy and Watergirl.

Cada jugador controlará desde una computadora diferente a Cappuccino Assassino o Ballerina Capuccina. Ambos personajes deberán colaborar para superar obstáculos, activar mecanismos, resolver puzles y escapar de una estructura misteriosa mediante el uso coordinado de sus habilidades especiales.

## Tecnologías principales

- Java JDK 17
- LibGDX 1.14.1
- Gradle
- IntelliJ IDEA
- Git y GitHub
- Plataforma objetivo: escritorio mediante LWJGL3

## Wiki del proyecto

La propuesta formal y detallada del videojuego se encuentra en la Wiki:

[Ver la propuesta del proyecto](https://github.com/agustinbruno07/capuccino-loves-ballerina/wiki)

## Compilación y ejecución

### Requisitos

Para compilar y ejecutar el proyecto es necesario contar con:

- Java JDK 17.
- Git.
- IntelliJ IDEA o cualquier entorno compatible con proyectos Gradle.

### Clonar el repositorio

Abrir una terminal y ejecutar:

```bash
git clone https://github.com/agustinbruno07/capuccino-loves-ballerina.git
```

### Ingresar a la carpeta del proyecto

Una vez clonado el repositorio, ejecutar:

```bash
cd capuccino-loves-ballerina
```

### Ejecutar en Windows

Desde la carpeta raíz del proyecto, ejecutar:

```bash
gradlew.bat lwjgl3:run
```

Si se utiliza Git Bash, también se puede ejecutar:

```bash
./gradlew.bat lwjgl3:run
```

### Ejecutar en Linux o macOS

Desde la carpeta raíz del proyecto, ejecutar:

```bash
./gradlew lwjgl3:run
```

### Ejecutar desde IntelliJ IDEA

1. Abrir IntelliJ IDEA.
2. Seleccionar `Open`.
3. Buscar y abrir la carpeta `capuccino-loves-ballerina`.
4. Esperar a que Gradle termine de importar el proyecto y descargar las dependencias.
5. Abrir el módulo `lwjgl3`.
6. Buscar la clase `Lwjgl3Launcher.java`.
7. Ejecutar el método `main` de `Lwjgl3Launcher`.

## [1.6.0] - 2026-11-07

### Agregado
- **Infraestructura de Red UDP:**
    - Clase `Servidor` y `HiloServidor`: Escucha puerto fijo, gestiona máx. 2 jugadores, asigna roles automáticos, mantiene estado autoritativo (palanca/puerta/cortes/muertes) y envía snapshots de bienvenida.
    - Clase `Cliente` y `HiloCliente`: Conexión asíncrona, cola de recepción segura (`synchronized`), métodos helpers para enviar inputs específicos (`enviarPosicion`, `enviarCorte`, etc.).
    - Soporte para configuración de IP del servidor mediante argumentos CLI en modo prueba.

### Corregido
- Bug crítico en `HiloServidor.mandarSnapshot`: Los flags booleanos (`mirando`, `enSuelo`) ahora se serializan como `"true"/"false"` consistentes con el broadcast en vivo, previniendo errores de parsing (`Boolean.parseBoolean("1.0") == false`) en clientes reconectados.

### Notas Técnicas
- Validación realizada mediante smoke test de 3 instancias locales simulando conexión secuencial y saturación de cupos.
- La integración final con `ControladorPartida` y `PantallaJuego` queda pendiente para la siguiente iteración.

### Agregado
- **Integración completa de Ballerina Capuccina:**

    - Visualización de Ballerina como rectángulo rosa con marcador direccional mientras se esperan sprites definitivos.

- Arte real de objetos interactivos: sprites independientes en `assets/objetos/` (puerta cerrada/abierta, palanca apagada/encendida, pinchos y liana) dibujados con tamaño fijo en el mundo, desacoplados de la resolución del PNG.
- Pinchos dibujados con proporción natural 2:1 apoyados sobre la zona letal; liana dibujada apilada en segmentos de 192px para evitar estiramiento.
- Clamp de delta en `render()` (máximo 1/30s) para evitar tunneling y movimiento errático en el primer frame tras la carga.

- `reintentar()` ahora también devuelve la puerta y la palanca a su estado inicial, además de reposicionar al personaje.
